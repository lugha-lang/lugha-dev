// Reaches for the internet, by name and by address: there is no network.
import std.http;

fun main(console: Console, net: Net) {
    console.println(http.get(net, "http://example.com/").is_ok());
    console.println(net.connect_timeout("1.1.1.1:53", 2000).is_ok());
    console.println(net.resolve("example.com").is_ok());
}
